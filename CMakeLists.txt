if(WDS_RUNTIME_PACK)
  if(WDS_LLAMA OR WDS_GGML_TRAIN OR GGML_CUDA)
    message(FATAL_ERROR "WDS_RUNTIME_PACK uses existing DLLs. Set WDS_LLAMA, WDS_GGML_TRAIN and GGML_CUDA OFF; the small Wind trainer uses native scalar SGD.")
  endif()
  target_compile_definitions(wds_core PRIVATE WDS_RUNTIME_PACK=1)
  if(WIN32)
    target_compile_definitions(wds_core PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  endif()
  target_include_directories(wds_core SYSTEM PRIVATE third_party/llama-b9870)
  target_sources(wds_core PRIVATE src/runtime_pack.cpp src/prompt.cpp)
  set(WDS_TEMPLATE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama-template")
  add_library(wds_jinja STATIC
    ${WDS_TEMPLATE_SOURCE}/jinja/lexer.cpp ${WDS_TEMPLATE_SOURCE}/jinja/parser.cpp
    ${WDS_TEMPLATE_SOURCE}/jinja/runtime.cpp ${WDS_TEMPLATE_SOURCE}/jinja/value.cpp
    ${WDS_TEMPLATE_SOURCE}/jinja/string.cpp ${WDS_TEMPLATE_SOURCE}/unicode.cpp ${WDS_TEMPLATE_SOURCE}/json.cpp)
  target_include_directories(wds_jinja PUBLIC ${WDS_TEMPLATE_SOURCE} PRIVATE third_party)
  target_link_libraries(wds_core PRIVATE wds_jinja ${CMAKE_DL_LIBS})
  message(STATUS "Using existing b9870 runtime pack; CUDA compiler is not enabled")
endif()


if(WDS_LLAMA OR WDS_GGML_TRAIN)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  if(WDS_LLAMA_SOURCE)
    if(WDS_LLAMA)
      add_subdirectory(${WDS_LLAMA_SOURCE} ${CMAKE_BINARY_DIR}/llama)
    else()
      add_subdirectory(${WDS_LLAMA_SOURCE}/ggml ${CMAKE_BINARY_DIR}/ggml)
    endif()
  else()
    include(FetchContent)
    if(WDS_LLAMA)
      FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
        GIT_TAG b1c902f026d32535b5e8eae55e5e47b379e57672)
    else()
      FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
        GIT_TAG b1c902f026d32535b5e8eae55e5e47b379e57672 SOURCE_SUBDIR ggml)
    endif()
    FetchContent_MakeAvailable(llama)
  endif()
  if(WDS_LLAMA)
    target_compile_definitions(wds_core PRIVATE WDS_LLAMA=1)
    # Reuse only the native template engine, not the common CLI/HTTP stack.
    if(WDS_LLAMA_SOURCE)
      set(WDS_JINJA_SOURCE "${WDS_LLAMA_SOURCE}")
    else()
      set(WDS_JINJA_SOURCE "${llama_SOURCE_DIR}")
    endif()
    add_library(wds_jinja STATIC
      ${WDS_JINJA_SOURCE}/common/jinja/lexer.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/parser.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/runtime.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/value.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/string.cpp
      ${WDS_JINJA_SOURCE}/common/unicode.cpp
      ${WDS_JINJA_SOURCE}/common/json.cpp)
    target_include_directories(wds_jinja PUBLIC ${WDS_JINJA_SOURCE}/common PRIVATE ${WDS_JINJA_SOURCE}/vendor)
    target_link_libraries(wds_jinja PRIVATE ggml-base)
    target_sources(wds_core PRIVATE src/prompt.cpp)
    target_link_libraries(wds_core PRIVATE llama wds_jinja)
  endif()
endif()
