cmake_minimum_required(VERSION 3.20)
project(wds_ai VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WDS_LLAMA "Build frozen llama.cpp foundation" OFF)
set(WDS_LLAMA_SOURCE "" CACHE PATH "Existing pinned llama.cpp checkout (optional)")
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)
find_package(Threads REQUIRED)
add_library(wds_core src/core.cpp src/winds.cpp src/intelligence.cpp src/relationships.cpp src/heart.cpp src/mesh.cpp src/tasks.cpp src/upkeep.cpp src/scopes.cpp src/workspace.cpp)
target_include_directories(wds_core PUBLIC include)
target_include_directories(wds_core SYSTEM PUBLIC third_party)
target_link_libraries(wds_core PUBLIC Threads::Threads)
add_library(wds.ai.understanding ALIAS wds_core)
option(WDS_GGML_TRAIN "Build small Wind optimizer with pinned ggml" OFF)
if(WDS_LLAMA OR WDS_GGML_TRAIN)
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  if(WDS_LLAMA_SOURCE)
    if(WDS_LLAMA)
      add_subdirectory(${WDS_LLAMA_SOURCE} ${CMAKE_BINARY_DIR}/llama)
    else()
      add_subdirectory(${WDS_LLAMA_SOURCE}/ggml ${CMAKE_BINARY_DIR}/ggml)
    endif()
  else()
    include(FetchContent)
    if(WDS_LLAMA)
      FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
        GIT_TAG b1c902f026d32535b5e8eae55e5e47b379e57672)
    else()
      FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
        GIT_TAG b1c902f026d32535b5e8eae55e5e47b379e57672 SOURCE_SUBDIR ggml)
    endif()
    FetchContent_MakeAvailable(llama)
  endif()
  if(WDS_LLAMA)
    target_compile_definitions(wds_core PRIVATE WDS_LLAMA=1)
    # Reuse only the native template engine, not the common CLI/HTTP stack.
    if(WDS_LLAMA_SOURCE)
      set(WDS_JINJA_SOURCE "${WDS_LLAMA_SOURCE}")
    else()
      set(WDS_JINJA_SOURCE "${llama_SOURCE_DIR}")
    endif()
    add_library(wds_jinja STATIC
      ${WDS_JINJA_SOURCE}/common/jinja/lexer.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/parser.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/runtime.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/value.cpp
      ${WDS_JINJA_SOURCE}/common/jinja/string.cpp
      ${WDS_JINJA_SOURCE}/common/unicode.cpp
      ${WDS_JINJA_SOURCE}/common/json.cpp)
    target_include_directories(wds_jinja PUBLIC ${WDS_JINJA_SOURCE}/common PRIVATE ${WDS_JINJA_SOURCE}/vendor)
    target_link_libraries(wds_jinja PRIVATE ggml-base)
    target_sources(wds_core PRIVATE src/prompt.cpp)
    target_link_libraries(wds_core PRIVATE llama wds_jinja)
  endif()
endif()
add_executable(wds-ai src/server.cpp)
target_include_directories(wds-ai SYSTEM PRIVATE third_party/asio)
target_compile_definitions(wds-ai PRIVATE ASIO_STANDALONE _WEBSOCKETPP_CPP11_STL_)
target_link_libraries(wds-ai PRIVATE wds_core)
if(WIN32)
  target_link_libraries(wds-ai PRIVATE ws2_32 mswsock)
endif()
if(MSVC)
  target_compile_options(wds_core PRIVATE /W4)
else()
  target_compile_options(wds_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(wds-tests tests/core_test.cpp)
target_link_libraries(wds-tests PRIVATE wds_core)
add_test(NAME core COMMAND wds-tests ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
add_executable(wds-heart-tests tests/heart_test.cpp)
target_link_libraries(wds-heart-tests PRIVATE wds_core)
add_test(NAME heart COMMAND wds-heart-tests ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME heart_websocket COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/heart_websocket_test.py $<TARGET_FILE:wds-ai> ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
  add_test(NAME websocket COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/websocket_test.py $<TARGET_FILE:wds-ai> ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
endif()
add_executable(wds-train src/trainer.cpp)
target_link_libraries(wds-train PRIVATE wds_core)
if(WDS_LLAMA OR WDS_GGML_TRAIN)
  target_compile_definitions(wds-train PRIVATE WDS_GGML_TRAIN=1)
  target_link_libraries(wds-train PRIVATE ggml ggml-cpu)
endif()
add_executable(wds-workshop-tests tests/workshop_test.cpp)
target_link_libraries(wds-workshop-tests PRIVATE wds_core)
add_test(NAME workshop_core COMMAND wds-workshop-tests ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)

if(Python3_FOUND)
  add_test(NAME native_trainer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/trainer_test.py $<TARGET_FILE:wds-train> $<TARGET_FILE:wds-ai> ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
  set_tests_properties(native_trainer PROPERTIES TIMEOUT 120)
endif()
add_executable(wds-upkeep-tests tests/upkeep_test.cpp)
target_link_libraries(wds-upkeep-tests PRIVATE wds_core)
add_test(NAME upkeep_core COMMAND wds-upkeep-tests ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
add_executable(wds-workspace-tests tests/workspace_test.cpp)
target_link_libraries(wds-workspace-tests PRIVATE wds_core)
add_test(NAME workspace_core COMMAND wds-workspace-tests ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
if(Python3_FOUND)
  add_test(NAME upkeep_websocket COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/upkeep_websocket_test.py $<TARGET_FILE:wds-ai> ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
  set_tests_properties(upkeep_websocket PROPERTIES TIMEOUT 90)
endif()
if(Python3_FOUND)
  add_test(NAME sixstars_setup COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/setup_test.py)
  add_test(NAME interface_websocket COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interface_test.py $<TARGET_FILE:wds-ai> ${CMAKE_CURRENT_SOURCE_DIR}/packages/breeze)
  set_tests_properties(interface_websocket PROPERTIES TIMEOUT 60)

  add_executable(wds-prompt-tests tests/prompt_test.cpp)
  target_link_libraries(wds-prompt-tests PRIVATE wds_core)
  add_test(NAME gemma4_template COMMAND wds-prompt-tests ${WDS_JINJA_SOURCE}/models/templates/google-gemma-4-31B-it.jinja)
